    end
end)

    local data = pcall(function() return game:GetService("HttpService"):JSONDecode(fileContent) end)
    -- Kiểm tra thời gian lưu (24 giờ = 86400 giây)
    if data and type(data) == "table" and data.Timestamp and (os.time() - data.Timestamp < 86400) then
        isKeyValid = true
    end
end

-- Hiệu ứng lắc mượt mà (Smooth Shake Effect)
local function smoothShake()
    local originalPos = UDim2.new(0.5, -200, 0.5, -115)
    local duration = 0.45
    local elapsed = 0
    local intensity = 10
    
    local connection
    connection = RunService.RenderStepped:Connect(function(dt)
        elapsed = elapsed + dt
        if elapsed < duration then
            local damping = 1 - (elapsed / duration)
            local offsetX = math.sin(elapsed * 45) * intensity * damping
            MainFrame.Position = originalPos + UDim2.new(0, offsetX, 0, 0)
        else
            MainFrame.Position = originalPos
            connection:Disconnect()
        end
    end)
end

-- Hiệu ứng rê chuột
local function applyEffect(btn)
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundTransparency = 0.2}):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundTransparency = 0}):Play()
    end)
end
applyEffect(SubmitBtn)
applyEffect(GetKeyBtn)
